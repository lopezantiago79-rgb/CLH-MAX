\begin{abstract}
We propose CLH-MAX, a Module-LWE-based key encapsulation mechanism (KEM) in the style of Kyber/ML-KEM in which the ring $\Z_q[X]/(X^{256}+1)$ is replaced by $\Rq=\Z_q[X]/\Phi_{257}(X)$, the ring of the prime cyclotomic field $\mathbb{Q}(\zeta_{257})$. For $q=3329$ the polynomial $\Phi_{257}$ is irreducible modulo $q$, so $\Rq\cong\F_{q^{256}}$ is a field. We specify the CPA-secure public-key encryption layer, its Fujisaki--Okamoto transform to an IND-CCA2 KEM (random-oracle model), and three parameter sets of module rank $2,3,4$. We give a game-hopping proof that IND-CPA security reduces to the decision Module-LWE problem over $\Rq$, an exact analysis of the decryption failure probability (which corrects the na\"ive coefficient-variance count for this ring), lattice-estimator outputs for the generic MLWE instances underlying the parameter sets (identical to those of ML-KEM-768/1024 for $k=3,4$, as expected, and about $7$ bits lower for $k=2$ when $\eta=2$ is used), and a discussion of arithmetic cost, where the absence of a negacyclic NTT is the main disadvantage. We make no claim of higher concrete security than ML-KEM; the contribution is a fully specified alternative ring instantiation together with its correctness and reduction analysis, reproducible scripts, and a clear list of open problems.
\end{abstract}

\section{Introduction}
ML-KEM (FIPS~203 \cite{FIPS203}) is based on Module-LWE over $\Z_{3329}[X]/(X^{256}+1)$, a power-of-two cyclotomic ring chosen because it admits a fast (incomplete) number-theoretic transform (NTT). The choice of ring in structured-lattice cryptography is a design parameter that is not fully explored: other cyclotomic rings have been proposed, e.g.\ prime-conductor rings in the ring-LWE literature and in NTRU Prime. This note studies one concrete alternative, conductor $257$ (the Fermat prime $F_3=2^8+1$), for which the ring degree is $\varphi(257)=256$, equal to that of ML-KEM, so module ranks, key sizes and noise parameters can be compared directly.

\paragraph{Contributions.}
(1) A complete specification of CLH-MAX: PKE, FO-based KEM, three parameter sets (Sections~\ref{sec:ring}--\ref{sec:scheme}). (2) A proof that IND-CPA security reduces to two decision-MLWE instances over $\Rq$, with the explicit simulators (Section~\ref{sec:sec}, Appendix~\ref{app:sim}). (3) An exact computation of the decryption failure probability including the ring-specific coefficient count for $\Phi_{257}$; it shows that the highest parameter set does \emph{not} reach $2^{-128}$ without modification (Section~\ref{sec:dfp}, Appendices~\ref{app:count},\ref{app:script}). (4) A cost analysis of multiplication in $\Rq$ (Section~\ref{sec:arith}).

\paragraph{Non-claims.} We do not claim that CLH-MAX is more secure than ML-KEM, that known attacks on ML-KEM fail to apply to a field ring, or that worst-case reductions give concrete security at these parameters. Section~\ref{sec:sec} explains why.

\paragraph{Related work.} Ring-LWE over general cyclotomic rings was introduced in \cite{LPR13}, including the worst-case reduction for arbitrary cyclotomic fields. Kyber \cite{BDKLLSSS18} and ML-KEM use module structure over $\Z_q[X]/(X^{256}+1)$; the module variant over $\Z_q[X]/\Phi_{257}$ studied here follows the same template. Subfield and structural attacks are discussed in \cite{ABD16,CJL16,CDPR16,CDW17}.

\section{Preliminaries}
\paragraph{Notation.} $N=256$, $q$ prime, $\Rq=\Z_q[X]/\Phi_{257}(X)$ with $\Phi_{257}=X^{256}+\dots+X+1$. Elements are identified with coefficient vectors in $\Z_q^{256}$. For $k\ge1$, vectors and matrices over $\Rq$ are written with the module rank $k$ as dimension. $\psi_\eta$ is the centered binomial distribution on $\Z$: $\sum_{i=1}^{\eta}(a_i-b_i)$ with $a_i,b_i\leftarrow\{0,1\}$ uniform; it has variance $\eta/2$. For $\eta=2$ the variance is $1$ (we use $\eta=2$ throughout, so $\sigma^2=1$). $\chi$ denotes $\psi_2$ applied coefficient-wise.

\begin{definition}[Decision MLWE]
Let $n,m\ge1$. $\mathrm{MLWE}_{n,m,\chi}$ asks to distinguish $(B,Bs+e)$ from $(B,w)$, where $B\leftarrow U(\Rq^{m\times n})$, $s\leftarrow\chi^n$, $e\leftarrow\chi^m$, $w\leftarrow U(\Rq^m)$. $\mathrm{Adv}^{\mathrm{MLWE}}_{n,m,\chi}(\mathcal D)$ is $|\Pr[\mathcal D(B,Bs+e)=1]-\Pr[\mathcal D(B,w)=1]|$.
\end{definition}

\begin{definition}[PKE, IND-CPA]
A PKE is $(\mathsf{KeyGen},\mathsf{Enc},\mathsf{Dec})$. In the IND-CPA game the challenger samples $(pk,sk)$ and a bit $b$; the adversary picks $m_0,m_1$ of equal length, receives $\mathsf{Enc}(pk,m_b)$ and outputs $b'$. The advantage is $|\Pr[b'=b]-\tfrac12|$.
\end{definition}

\begin{definition}[KEM, IND-CCA2]
A KEM is $(\mathsf{KeyGen},\mathsf{Encaps},\mathsf{Decaps})$, with $\mathsf{Encaps}(pk)\to(c,K)$. In the IND-CCA2 game the adversary receives $(c^*,K_b)$ with $K_0$ the real key and $K_1$ uniform, and may query $\mathsf{Decaps}$ on any $c\neq c^*$; advantage $|\Pr[b'=b]-\tfrac12|$.
\end{definition}

\section{The ring}\label{sec:ring}
\begin{proposition}\label{prop:irr}
Let $q\ne257$ be prime. Then $\Phi_{257}$ is irreducible over $\F_q$ if and only if $\mathrm{ord}_{257}(q)=256$, i.e.\ $q$ is a primitive root modulo $257$. In that case $\Rq\cong\F_{q^{256}}$ and $q\mathcal{O}_K$ is a prime ideal of $\mathcal{O}_K=\Z[\zeta_{257}]$.
\end{proposition}
\begin{proof}
For $\gcd(q,m)=1$ all irreducible factors of $\Phi_m$ over $\F_q$ have degree $\mathrm{ord}_m(q)$; here $\deg\Phi_{257}=\varphi(257)=256$. The statement about $q\mathcal O_K$ is the standard description of inert primes in cyclotomic fields.
\end{proof}
Direct computation gives $\mathrm{ord}_{257}(3329)=\mathrm{ord}_{257}(12289)=256$ ($3329\equiv245$ and $12289\equiv210 \pmod{257}$). Among primes $3000<q<20000$ there are $922$ moduli with this property (e.g.\ $3001,3019,3037,3041,\dots$). Not every NTT-friendly prime works: $\mathrm{ord}_{257}(7681)=128$, so $\Phi_{257}$ splits into two factors of degree $128$ modulo $7681$. We use $q=3329$, the ML-KEM modulus.

\begin{remark}[Comparison with ML-KEM]
For $q=3329$ one has $q\equiv 1 \pmod{256}$ but $q\not\equiv 1\pmod{512}$ ($3329\bmod 512=257$, $\mathrm{ord}_{512}(3329)=2$). Hence $X^{256}+1$ factors modulo $3329$ into $128$ \emph{quadratic} factors, not $256$ linear ones; this is why ML-KEM uses an incomplete, 7-layer NTT. $q=12289$, which splits $X^{256}+1$ completely, is not an ML-KEM modulus.
\end{remark}

\begin{remark}[Subfields]\label{rem:sub}
$\F_{q^{256}}$ is not ``subfield-free'': it contains the subfields $\F_{q^{2^j}}$, $0\le j\le 7$, and $\mathrm{Gal}(\F_{q^{256}}/\F_q)$ is cyclic of order $256$; likewise $\mathbb{Q}(\zeta_{257})$ has a cyclic Galois group of order $256$ and a subfield of every degree $2^j$. What differs from $X^{256}+1$ is only that $\Rq$ does not decompose as a product of smaller rings. Whether this has any security consequence is open.
\end{remark}

\paragraph{Multiplication.} Let $a,b\in\Rq$ with coefficient vectors of length $256$ (degree $\le255$). Then $ab=\sum_{i=0}^{510}c_iX^i$ over $\Z_q$. Since $\Phi_{257}\mid X^{257}-1$, first fold $p_i=\sum_{j+l\equiv i\ (257)}a_jb_l$ for $0\le i\le256$; then $ab\bmod\Phi_{257}$ has coefficients $p_i-p_{256}$ for $0\le i\le255$ (because $X^{256}\equiv-(X^{255}+\dots+1)$). Hence a ring multiplication is a length-$257$ cyclic convolution followed by $256$ subtractions.

\section{Scheme}\label{sec:scheme}
Fix $q=3329$, $\eta=2$, module rank $k\in\{2,3,4\}$. Let $\mathrm{Enc}_1(m)=\lceil q/2\rfloor\cdot m$ embed $m\in\{0,1\}^{256}$ coefficient-wise into $\Rq$ and $\mathrm{Dec}_1(x)_i=1$ iff the centered representative of $x_i$ lies in $[\lceil q/4\rceil,\lfloor 3q/4\rfloor]$.

\paragraph{PKE.}
\begin{itemize}[nosep]
\item $\mathsf{KeyGen}$: $A\leftarrow U(\Rq^{k\times k})$; $s,e\leftarrow\chi^k$; $t=As+e$. $pk=(A,t)$, $sk=s$.
\item $\mathsf{Enc}(pk,m;r)$: coins determine $r,e_1\leftarrow\chi^k$, $e_2\leftarrow\chi$; $u=A^{T}r+e_1$; $v=t^{T}r+e_2+\mathrm{Enc}_1(m)$. Output $(u,v)$.
\item $\mathsf{Dec}(sk,(u,v))$: $m'=\mathrm{Dec}_1(v-s^{T}u)$.
\end{itemize}
\textbf{Correctness.} $v-s^{T}u=\mathrm{Enc}_1(m)+E$ with $E=e^{T}r+e_2-s^{T}e_1$; decryption succeeds iff every coefficient of $E$ has centered absolute value $<\lfloor q/4\rfloor$ (up to rounding of $q/2$).

\paragraph{KEM (FO with implicit rejection).} With hash functions $H,G$ modelled as random oracles and following \cite{HHK17}:
\begin{itemize}[nosep]
\item $\mathsf{KeyGen}$: $(pk,s)\leftarrow\mathsf{PKE.KeyGen}$, $z\leftarrow\{0,1\}^{256}$; $sk_{\mathrm{KEM}}=(s,pk,H(pk),z)$.
\item $\mathsf{Encaps}(pk)$: $m\leftarrow\{0,1\}^{256}$; $(\bar K,r)=G(m\,\|\,H(pk))$; $c=\mathsf{Enc}(pk,m;r)$; output $(c,\,K=H(\bar K\,\|\,H(c)))$.
\item $\mathsf{Decaps}(sk,c)$: $m'=\mathsf{Dec}(s,c)$; $(\bar K',r')=G(m'\|H(pk))$; if $c=\mathsf{Enc}(pk,m';r')$ (constant-time comparison) return $H(\bar K'\|H(c))$, else return $H(z\|H(c))$ (implicit rejection, also in constant time).
\end{itemize}
By \cite{HHK17}, IND-CCA2 security of this KEM in the random-oracle model follows from IND-CPA security of the PKE together with a sufficiently small decryption failure probability $\delta$; the explicit bound of \cite{HHK17} should be consulted for the concrete loss in terms of the number of oracle queries. We make no quantum-random-oracle claim.

\paragraph{Sizes (no compression).} With $12$-bit packing: $|pk|=384k+32$ B, $|c|=384(k+1)$ B, $|sk_{\mathrm{KEM}}|=384k+|pk|+64$ B.

\section{Security}\label{sec:sec}
\begin{theorem}[IND-CPA]\label{thm:cpa}
For every adversary $\mathcal A$ against the PKE there are distinguishers $\mathcal B_1,\mathcal B_2$ with essentially the running time of $\mathcal A$ such that
\[
\mathrm{Adv}^{\mathrm{IND\text{-}CPA}}(\mathcal A)\le \mathrm{Adv}^{\mathrm{MLWE}}_{k,k,\chi}(\mathcal B_1)+\mathrm{Adv}^{\mathrm{MLWE}}_{k+1,k,\chi}(\mathcal B_2).
\]
\end{theorem}
\begin{proof}
\emph{Game $G_0$} is the real IND-CPA game. \emph{Game $G_1$} is $G_0$ with $t$ replaced by a uniform $t^*\in\Rq^k$. $\mathcal B_1$, given an MLWE challenge $(A,t^*)$ with $k$ samples and secret $s$, runs $\mathcal A$ on $pk=(A,t^*)$, plays the challenger honestly for the rest and outputs $1$ iff $\mathcal A$ wins. If $t^*=As+e$ this is exactly $G_0$, otherwise $G_1$, so $|\Pr[G_0]-\Pr[G_1]|=\mathrm{Adv}^{\mathrm{MLWE}}_{k,k,\chi}(\mathcal B_1)$.
\emph{Game $G_2$} replaces $(u,v)$ by uniform elements of $\Rq^k\times\Rq$. In $G_1$ the matrix $B=\binom{A^{T}}{t^{T}}\in\Rq^{(k+1)\times k}$ is uniform and
\[
\binom{u}{v-\mathrm{Enc}_1(m_b)}=B\,r+\binom{e_1}{e_2}
\]
is an MLWE sample with secret $r$ and $k+1$ rows. $\mathcal B_2$, given $(B,w)$, parses $B$ into $A^{T}$ and $t^{T}$, runs $\mathcal A$ on $pk=(A,t)$, and after the challenge messages are chosen returns $\mathcal A$ the ciphertext $(w_{1..k},\,w_{k+1}+\mathrm{Enc}_1(m_b))$. Real samples give $G_1$, uniform ones $G_2$. In $G_2$ the ciphertext is independent of $b$ so $\Pr[G_2]=\tfrac12$. Combining, $\Pr[G_0]-\tfrac12\le\varepsilon_1+\varepsilon_2$.
\end{proof}
\begin{remark}[Why no ``axiom'']
The theorem is a relative statement. It does \emph{not} give $\mathrm{Adv}^{\mathrm{IND\text{-}CPA}}=0$: an unbounded adversary recovers $s$ from $(A,t)$ (the support of $(s,e)$ is far smaller than $q^{256k}$), so no sound assumption can set the MLWE advantage of \emph{all} distinguishers to zero. Security is computational and rests on the hardness of MLWE for efficient adversaries. Any mechanised proof should formalise the inequality of Theorem~\ref{thm:cpa}, not an unconditional-hardness axiom.
\end{remark}

\paragraph{Worst-case reductions.} Lyubashevsky--Peikert--Regev \cite{LPR13} give a quantum reduction from approximate worst-case SVP on ideal lattices to RLWE for \emph{any} cyclotomic field, hence including $\mathbb{Q}(\zeta_{257})$, but for discrete Gaussian errors in the canonical embedding with width $\alpha q\ge\omega(\sqrt{\log n})$ and with moduli and approximation factors that are polynomial in $n$ with non-trivial exponents. The parameters used here ($q=3329$, centered binomial $\eta=2$, $n=256$) lie outside that regime, exactly as for ML-KEM, so the reductions provide asymptotic motivation only and no concrete bits of security. Moreover, for prime-power cyclotomic fields (which includes conductor $257$) there are quantum algorithms for ideal-SVP with approximation factors $\exp(\tilde O(\sqrt n))$ in certain settings \cite{CDPR16,CDW17}; these do not break RLWE at the parameters considered, but they caution against reading worst-case statements as guarantees.

\paragraph{Concrete security.} Attacks on (M)LWE (primal uSVP, BDD, dual, hybrid) as implemented in the lattice-estimator \cite{APS15} see only the triple $(n,q,\chi)$ with $n=kN$; they do not depend on the ring structure. We ran \texttt{LWE.estimate} (default settings, October 2026) with $m=n$, $q=3329$ on the key-recovery instance of each CLH-MAX parameter set and, with the \emph{same call}, on the corresponding ML-KEM instance. Table~\ref{tab:est} reports the output unchanged.

The outcome is what the structure of the estimator predicts: for $k=3,4$ the instances coincide with ML-KEM-768/1024 and so do the estimates; for $k=2$ the choice $\eta=2$ makes CLH-MAX-512 about $7$ bits \emph{weaker} than ML-KEM-512 (which uses $\psi_3$ for the key). Known structural attacks (subfield attacks on overstretched NTRU \cite{ABD16,CJL16}) target large-modulus NTRU-type instances and are not known to apply to MLWE at $q=3329$ in either ring; Remark~\ref{rem:sub} explains why the field structure of $\Rq$ is not obviously free of exploitable structure.

\section{Correctness: decryption failure probability}\label{sec:dfp}
\begin{lemma}\label{lem:count}
For $a,b\in\Rq$ with independent coefficients, every coefficient of $ab\in\Rq$ is a signed sum of $510$ products $a_jb_l$ of distinct index pairs, except coefficient $255$ which is a signed sum of $511$.
\end{lemma}
\noindent The proof is a direct count of index pairs (Appendix~\ref{app:count}). This is roughly twice the $256$ terms of a negacyclic product, so the variance of a product coefficient is about $2\sigma^4\cdot256$ rather than $256\sigma^4$.

For module rank $k$, the coefficient $E_i$ of $e^{T}r-s^{T}e_1+e_2$ is therefore a sum of $2k\cdot511$ products of two independent $\psi_2$ samples (a conservative count) plus one $\psi_2$ sample, with variance $\sigma_E^2=1022k+1$. Following the usual heuristic of treating the summands as independent, we compute the \emph{exact} distribution by convolution and bound $\delta\le 256\cdot\Pr[|E_i|\ge\lfloor q/4\rfloor]$ with $\lfloor q/4\rfloor=832$. The script is in Appendix~\ref{app:script}.

Sets with $k=2,3$ satisfy $\delta<2^{-128}$; \textbf{the set with $k=4$ does not} ($\delta\approx2^{-117}$). Using it inside FO would require changing parameters (e.g.\ a larger modulus with adapted noise, or an error-correcting encoding of $m$), which would also change the security analysis; we leave this open. The Gaussian tail approximation $\exp(-832^2/2\sigma_E^2)$ gives $\approx 2^{-245}$ per coefficient for $k=2$, in line with the exact value.

\paragraph{Trade-off between security and correctness at $k=2$.} Raising the key noise of CLH-MAX-512 to ML-KEM-512's $\eta_1=3$ (with $r\leftarrow\psi_3$, $e_1,e_2\leftarrow\psi_2$, so that the key-recovery instance equals the ML-KEM-512 row of Table~\ref{tab:est}) closes the $7$-bit gap but raises $\sigma_E^2$ to about $3834$; the exact computation gives $\delta\approx2^{-123}$, which does \emph{not} meet $2^{-128}$ (ML-KEM-512 achieves its failure rate with only $256$ terms per product coefficient, plus ciphertext compression effects not modelled here). Thus for conductor $257$ the near doubling of terms per coefficient (Lemma~\ref{lem:count}) forces a choice at $k=2$: $\eta=2$ with $\delta\approx2^{-231}$ and about $2^{132.8}$ (dual hybrid), or $\eta_1=3$ with $\delta\approx2^{-123}$ and about $2^{139.7}$. We specify the former and regard the latter as an open design question.

\section{Arithmetic}\label{sec:arith}
By Section~\ref{sec:ring}, multiplication in $\Rq$ is a length-$257$ cyclic convolution followed by a correction. Schoolbook multiplication costs $256^2=65{,}536$ coefficient products per ring multiplication; $\mathsf{KeyGen}$ needs $k^2$ ring products for $As$ and $\mathsf{Enc}$ needs $k^2+k+k$ more. A fast NTT route requires a transform of length $\ge511$ covering the linear convolution, i.e.\ $512\mid q-1$. $q=12289$ satisfies this ($12288=3\cdot2^{12}$) and $\mathrm{ord}_{257}(12289)=256$, but $q=3329$ does not ($3328=2^8\cdot13$). With $q=3329$ one must use Karatsuba/Toom--Cook, Bluestein/Rader-type or lifted-modulus methods. No implementation or benchmark is presented and we make no performance claim. Changing to $q=12289$ would require re-deriving the noise parameters, the security estimates and $\delta$.

\section{Open problems}
(i) Choose $k=2$ and $k=4$ parameters that meet both $\delta<2^{-128}$ and ML-KEM-level security (the extra terms per product coefficient in $\Rq$ cost correctness; ciphertext compression, an error-correcting message encoding or a different modulus are candidates). \section{Proof of Lemma~\ref{lem:count}}\label{app:count}
Coefficient $i$ of $ab\bmod\Phi_{257}$ is $p_i-p_{256}$ with $p_i=\sum_{j+l\equiv i\ (257)}a_jb_l$, $0\le j,l\le255$. For $0\le i\le255$: pairs with $j+l=i$ number $i+1$, and pairs with $j+l=i+257$ number $255-(i+2)+1=254-i$ (for $i\le254$) and $0$ for $i=255$. So $p_i$ has $255$ terms for $i\le254$ and $256$ terms for $i=255$. For $p_{256}$: $j+l=256$ gives $255$ pairs ($j=1,\dots,255$); $j+l=513>510$ is impossible, so $p_{256}$ has $255$ terms. The index-pair sets of $p_i$ and $p_{256}$ are disjoint, hence the coefficient is a signed sum of $255+255=510$ products, or $256+255=511$ for $i=255$. $\square$

\section{Simulators for Theorem~\ref{thm:cpa}}\label{app:sim}

B1(A, t*):                       # MLWE challenge: k samples, secret s

  b' <- Adv.guess(pk, (u, v)); return [b' == b]

B2(B, w):                        # MLWE challenge: k+1 samples, secret r
  parse B = [A^T ; t^T]          # A in R^{k x k}, t in R^k
  (m0, m1) <- Adv.choose(pk=(A, t)); b <- {0,1}
  u = w[1..k]; v = w[k+1] + Enc1(m_b)
  b' <- Adv.guess(pk, (u, v)); return [b' == b]
\end{verbatim}

\paragraph{Decryption failure probability} (exact convolution under the independence heuristic; $e,s,r\leftarrow\psi_{\eta_1}$, $e_1,e_2\leftarrow\psi_{\eta_2}$).
\begin{verbatim}
import numpy as np
from math import log2, comb
def cbd(eta):
    return np.array([comb(2*eta, i) for i in range(2*eta+1)], float)/4**eta, eta
def prod_pmf(x, y):
    p1, o1 = cbd(x); p2, o2 = cbd(y); d = {}
    for i, a in enumerate(range(-o1, o1+1)):
        for j, b in enumerate(range(-o2, o2+1)):
            d[a*b] = d.get(a*b, 0) + p1[i]*p2[j]
    off = max(abs(v) for v in d); arr = np.zeros(2*off+1)
    for v, p in d.items(): arr[v+off] += p
    return arr, off
def conv_pow(p, off, m):
    res, roff, base, boff = np.array([1.0]), 0, p.copy(), off
    while m:
        if m & 1: res = np.convolve(res, base); roff += boff
        m >>= 1
        if m: base = np.convolve(base, base); boff *= 2
    return res, roff
def delta(k, eta1, eta2, thr=832):        # thr = floor(q/4), q = 3329
    pa, oa = prod_pmf(eta1, eta1); pb, ob = prod_pmf(eta1, eta2)
    d1, o1 = conv_pow(pa, oa, k*511); d2, o2 = conv_pow(pb, ob, k*511)
    d, o = np.convolve(d1, d2), o1 + o2
    p2, oo = cbd(eta2); d, o = np.convolve(d, p2), o + oo
    idx = np.arange(len(d)) - o
    t = d[np.abs(idx) >= thr].sum()
    return log2(t), log2(t) + 8           # per coefficient, union over 256
for k, e1, e2 in [(2,2,2), (3,2,2), (4,2,2), (2,3,2)]:
    print(k, e1, e2, delta(k, e1, e2))
\end{verbatim}
